1. git init - ініціалізація нового Git-репозиторію.
2. git clone - клонування існуючого віддаленого репозиторію на локальний комп'ютер.
3. git status - перевірка поточного стану репозиторію (змінені файли, поточна гілка).
4. git add - додавання файлів до зони підготовки (staging area) для наступного коміту.
5. git commit - збереження поточного стану файлів (фіксація змін) у репозиторії.
6. git push - відправка локальних комітів до віддаленого репозиторію (наприклад, на GitHub).
7. git pull - отримання та об'єднання змін з віддаленого репозиторію у локальний.
8. git branch - створення, перегляд або видалення гілок.
9. git checkout або git switch - перемикання між гілками або відновлення файлів.
10. git merge - об'єднання змін з однієї гілки в іншу.
11. git log - перегляд історії комітів.
